DDL for DBLPScheme

These are the declarations in the ARANEUS Data Definition Language for a portion of DBLPScheme, the ADM scheme for the DB and LP Web Server

Lines starting with the symbol # are commented. The corresponding attributes are not available. 

SCHEME DBLPScheme http://www.informatik.uni-trier.de


  PAGE-SCHEME DBLPHomePage  UNIQUE /~ley/db/index.html
        Authors : LINK-TO AuthorSearchPage;
        Titles : LINK-TO TitleSearchPage;
  END                   



 PAGE-SCHEME AuthorSearchPage UNIQUE  /~ley/db/indices/a-tree/index.html
        NameForm : FORM(
                         Author : TEXT ;   
                         Submit : LINK-TO AuthorPage UNION AuthorIndexPage ;
                        );
 END                    

 PAGE-SCHEME TitleSearchPage UNIQUE  /~ley/db/indices/t-form.html
        TitleForm : FORM(
                          Keywords : TEXT ;   
                          Submit   : LINK-TO TitleIndexPage;
                        );
 END                    


 PAGE-SCHEME AuthorIndexPage 
        AuthorList: LIST-OF ( Name         : TEXT;
                              ToAuthorPage : LINK-TO AuthorPage ;
                            );
 END


 PAGE-SCHEME TitleIndexPage 
        WorkList: LIST-OF ( Authors   : TEXT;
                            Title     : TEXT;
                            Reference : TEXT;
                            Year      : TEXT;
                            Pages     : TEXT;
                            AuthorList: LIST-OF ( Name         : TEXT;
                                                  ToAuthorPage : LINK-TO AuthorPage ;
                                                );
                            ToRefPage : LINK-TO ConferencePage UNION JournalPage UNION SeriesPage UNION WebPage OPTIONAL;
                        );
 END


 PAGE-SCHEME AuthorPage 
        Name    : TEXT;
        HomePage: LINK-TO WebPage;
        WorkList: LIST-OF ( Authors   : TEXT;
                            Title     : TEXT;
                            Reference : TEXT;
                            Year      : TEXT;
                            Pages     : TEXT;
                            AuthorList: LIST-OF ( Name         : TEXT;
                                                  ToAuthorPage : LINK-TO AuthorPage ;
                                                );
                            ToRefPage : LINK-TO ConferencePage UNION JournalPage UNION SeriesPage  UNION WebPage OPTIONAL;
                            );
 END



 PAGE-SCHEME ConferencePage
        Conference : TEXT;
        Year       : TEXT;
        Location   : TEXT;
        PaperList  : LIST-OF ( Authors   : TEXT;
                               Title     : TEXT;
                               AuthorList: LIST-OF ( Name         : TEXT;
                                                     ToAuthorPage : LINK-TO AuthorPage ;
                                                   );
                             );
 END



 PAGE-SCHEME JournalPage
        Journal    : TEXT;
        Volume     : TEXT;
        Year       : TEXT;
        PaperList  : LIST-OF ( Authors   : TEXT;
                               Title     : TEXT;
                               Number    : TEXT;
                               AuthorList: LIST-OF ( Name         : TEXT;
                                                     ToAuthorPage : LINK-TO AuthorPage ;
                                                   );
                             );
 END



 PAGE-SCHEME SeriesPage
        Title  : TEXT;
        Number : TEXT;
        PaperList  : LIST-OF ( Authors   : TEXT;
                               Title     : TEXT;
                               AuthorList: LIST-OF ( Name         : TEXT;
                                                     ToAuthorPage : LINK-TO AuthorPage OPTIONAL ;
                                                   );
                             );
 END


 PAGE-SCHEME WebPage 
        Title    : TEXT;
        Type     : TEXT;
        Length   : TEXT;
        Modif    : TEXT;
        LinkList : LIST-OF ( 
                             Label    : TEXT OPTIONAL;
                             LinkType : TEXT;
                             HRef     : LINK-TO WebPage;
                           );
 END


END-SCHEME
