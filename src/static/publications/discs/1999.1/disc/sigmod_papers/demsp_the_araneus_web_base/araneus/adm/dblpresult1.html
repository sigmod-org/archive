View Example:

"Title, Authors, Reference and Year of all Papers on the World Wide Web"

DEFINE TABLE WWWPapers (Authors, Title, Reference, Year
AS           TitleSearchPage.TitleForm.Submit ->
             TitleIndexPage.WorkList
IN           DBLPScheme
USING        TitleIndexPage.WorkList.Authors,
             TitleIndexPage.WorkList.Title,
             TitleIndexPage.WorkList.Reference,
             TitleIndexPage.WorkList.Year       
WHERE        TitleSearchPage.TitleForm.Keywords = 'World Wide Web'
