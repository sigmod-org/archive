View Example:

"Name, Location and Year of Conferences in which Codd has published a paper after 1974"

DEFINE TABLE ConferencesCoddAfter74  (Conference, Location, Year)
AS           AuthorSearchPage.NameForm.Submit ->
             AuthorPage.WorkList.ToRefPage->
             ConferencePage
IN           DBLPScheme
USING        ConferencePage.Conference,
             ConferencePage.Location,
             ConferencePage.Year
WHERE        AuthorSearchPage.NameForm.Name = 'E. F. Codd',
             AuthorPage.WorkList.Year > '1974'
