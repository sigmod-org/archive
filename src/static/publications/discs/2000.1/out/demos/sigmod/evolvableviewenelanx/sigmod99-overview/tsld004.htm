    CREATE VIEW PotentialCustomer (VE = SUPERSET ) AS
    SELECT IS2.MASSBRANCH.CName ( AD = FALSE, AR = TRUE )
    IS2.MASSBRANCH.Phone ( AD = TRUE , AR = TRUE )
    FROM IS2.MASSBRANCH ( RD = FALSE, RR = TRUE )
    IS3.FLIGHTRES ( RD = FALSE, RR = FALSE )
    WHERE IS2.MASSBRANCH.CName=IS3.FLIGHTRES.PName ( CD = FALSE, CR = TRUE )
